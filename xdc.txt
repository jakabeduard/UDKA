-- Master XDC File for Nexys A7 with 7-segment display

## Clock Signal
set_property -dict { PACKAGE_PIN E3   IOSTANDARD LVCMOS33 } [get_ports clk]
create_clock -add -name sys_clk_pin -period 10.00 [get_ports clk]

## Reset Signal
set_property -dict { PACKAGE_PIN C12  IOSTANDARD LVCMOS33 } [get_ports rst]

## Start Signal
set_property -dict { PACKAGE_PIN D12  IOSTANDARD LVCMOS33 } [get_ports start]

## Stop Signal
set_property -dict { PACKAGE_PIN E16  IOSTANDARD LVCMOS33 } [get_ports stop]

## SPI Signals
set_property -dict { PACKAGE_PIN F13  IOSTANDARD LVCMOS33 } [get_ports sclk]
set_property -dict { PACKAGE_PIN G13  IOSTANDARD LVCMOS33 } [get_ports cs]
set_property -dict { PACKAGE_PIN H16  IOSTANDARD LVCMOS33 } [get_ports mosi1]
set_property -dict { PACKAGE_PIN J16  IOSTANDARD LVCMOS33 } [get_ports mosi2]
set_property -dict { PACKAGE_PIN K16  IOSTANDARD LVCMOS33 } [get_ports miso]

set_property -dict { PACKAGE_PIN V5   IOSTANDARD LVCMOS33 } [get_ports {seg_data[4]}]
set_property -dict { PACKAGE_PIN U5   IOSTANDARD LVCMOS33 } [get_ports {seg_data[5]}]
set_property -dict { PACKAGE_PIN V6   IOSTANDARD LVCMOS33 } [get_ports {seg_data[6]}]
